import re
import csv
import sys
from pathlib import Path


# =========================================================
# BASIC HELPERS
# =========================================================

def normalize_space(text):
    return re.sub(r"\s+", " ", text).strip()


def clean_sql(sql):
    """
    Removes SQL comments but keeps the SQL structure.
    """

    # Remove /* ... */
    sql = re.sub(
        r"/\*.*?\*/",
        " ",
        sql,
        flags=re.DOTALL
    )

    # Remove -- comments
    sql = re.sub(
        r"--.*?$",
        " ",
        sql,
        flags=re.MULTILINE
    )

    return sql


def clean_identifier(name):
    """
    Removes common quote characters from identifiers.
    """

    if not name:
        return ""

    return (
        name
        .replace('"', '')
        .replace("`", "")
        .replace("[", "")
        .replace("]", "")
        .strip()
    )


# =========================================================
# FIND MATCHING PARENTHESIS
# =========================================================

def find_matching_parenthesis(text, open_pos):
    """
    Given position of '(' find matching ')'.

    Handles nested parentheses and quoted strings.
    """

    depth = 0

    in_single_quote = False
    in_double_quote = False

    i = open_pos

    while i < len(text):

        char = text[i]

        # Handle single quotes
        if char == "'" and not in_double_quote:

            # SQL escaped quote ''
            if (
                in_single_quote
                and i + 1 < len(text)
                and text[i + 1] == "'"
            ):
                i += 2
                continue

            in_single_quote = not in_single_quote

        # Handle double quotes
        elif char == '"' and not in_single_quote:
            in_double_quote = not in_double_quote

        elif not in_single_quote and not in_double_quote:

            if char == "(":
                depth += 1

            elif char == ")":
                depth -= 1

                if depth == 0:
                    return i

        i += 1

    return -1


# =========================================================
# EXTRACT TOP-LEVEL CTEs
# =========================================================

def extract_cte_blocks(sql):
    """
    Extracts top-level CTEs.

    Returns:

    [
        {
            "name": "credit_txn_base",
            "body": "SELECT ...",
            "start": 100,
            "end": 500
        }
    ]
    """

    results = []

    # Find WITH
    with_match = re.search(
        r"\bWITH\b",
        sql,
        re.IGNORECASE
    )

    if not with_match:
        return results

    pos = with_match.end()

    while pos < len(sql):

        # Skip whitespace and comma
        while (
            pos < len(sql)
            and (
                sql[pos].isspace()
                or sql[pos] == ","
            )
        ):
            pos += 1

        # Match CTE name
        name_match = re.match(
            r'(["`\[]?[A-Za-z_][A-Za-z0-9_$]*["`\]]?)',
            sql[pos:]
        )

        if not name_match:
            break

        cte_name = clean_identifier(
            name_match.group(1)
        )

        cte_name_start = pos

        pos += name_match.end()

        # Optional column list:
        # cte_name(col1,col2) AS (...)
        while pos < len(sql) and sql[pos].isspace():
            pos += 1

        if pos < len(sql) and sql[pos] == "(":

            column_end = find_matching_parenthesis(
                sql,
                pos
            )

            if column_end == -1:
                break

            pos = column_end + 1

        # Skip whitespace
        while pos < len(sql) and sql[pos].isspace():
            pos += 1

        # Must have AS
        as_match = re.match(
            r"AS\b",
            sql[pos:],
            re.IGNORECASE
        )

        if not as_match:
            break

        pos += as_match.end()

        while pos < len(sql) and sql[pos].isspace():
            pos += 1

        if pos >= len(sql) or sql[pos] != "(":
            break

        body_start_paren = pos

        body_end_paren = find_matching_parenthesis(
            sql,
            body_start_paren
        )

        if body_end_paren == -1:
            break

        body = sql[
            body_start_paren + 1:
            body_end_paren
        ]

        results.append({
            "name": cte_name,
            "body": body,
            "start": cte_name_start,
            "end": body_end_paren
        })

        pos = body_end_paren + 1

        # After a CTE:
        # either ", next_cte AS (...)"
        # or final SELECT

        temp = pos

        while temp < len(sql) and sql[temp].isspace():
            temp += 1

        if temp < len(sql) and sql[temp] == ",":
            pos = temp + 1
            continue

        break

    return results


# =========================================================
# EXTRACT FROM/JOIN REFERENCES FROM A CTE
# =========================================================

def extract_references(cte_body):
    """
    Finds FROM and JOIN references.

    Example:

    FROM schema.table t
    LEFT JOIN Customer_Dim c
    """

    pattern = re.compile(
        r"\b(FROM|JOIN)\s+"
        r'(["`\[\]A-Za-z0-9_$#.]+)'
        r"(?:\s+(?:AS\s+)?"
        r"([A-Za-z_][A-Za-z0-9_$]*))?",
        re.IGNORECASE
    )

    invalid_aliases = {
        "WHERE",
        "LEFT",
        "RIGHT",
        "INNER",
        "FULL",
        "OUTER",
        "JOIN",
        "ON",
        "GROUP",
        "ORDER",
        "UNION",
        "HAVING",
        "CROSS",
        "LIMIT",
        "QUALIFY"
    }

    results = []

    for match in pattern.finditer(cte_body):

        source_type = match.group(1).upper()

        obj = clean_identifier(
            match.group(2)
        )

        alias = match.group(3) or ""

        if alias.upper() in invalid_aliases:
            alias = ""

        # Ignore accidental SQL keywords
        if obj.upper() in {
            "SELECT",
            "WHERE",
            "GROUP",
            "ORDER"
        }:
            continue

        results.append({
            "source_type": source_type,
            "object_name": obj,
            "alias": alias
        })

    return results


# =========================================================
# EXTRACT JOINS WITH CONDITIONS
# =========================================================

def extract_joins(cte_body):
    """
    Extract joins and ON conditions inside one CTE.
    """

    pattern = re.compile(
        r"\b("
        r"LEFT(?:\s+OUTER)?|"
        r"RIGHT(?:\s+OUTER)?|"
        r"FULL(?:\s+OUTER)?|"
        r"INNER|"
        r"CROSS"
        r")?\s*JOIN\s+"
        r'(["`\[\]A-Za-z0-9_$#.]+)'
        r"(?:\s+(?:AS\s+)?"
        r"([A-Za-z_][A-Za-z0-9_$]*))?"
        r"(?:\s+ON\s+(.*?))?"
        r"(?="
        r"\b(?:LEFT|RIGHT|FULL|INNER|CROSS)"
        r"(?:\s+OUTER)?\s+JOIN\b|"
        r"\bJOIN\b|"
        r"\bWHERE\b|"
        r"\bGROUP\s+BY\b|"
        r"\bHAVING\b|"
        r"\bQUALIFY\b|"
        r"\bUNION\b|"
        r"\bORDER\s+BY\b|"
        r"$"
        r")",
        re.IGNORECASE | re.DOTALL
    )

    results = []

    for match in pattern.finditer(cte_body):

        join_type = match.group(1)

        if join_type:
            join_type = normalize_space(
                join_type
            ).upper()
        else:
            join_type = "INNER"

        object_name = clean_identifier(
            match.group(2)
        )

        alias = match.group(3) or ""

        condition = match.group(4) or ""

        results.append({
            "join_type": join_type,
            "object_name": object_name,
            "alias": alias,
            "condition": normalize_space(
                condition
            )
        })

    return results


# =========================================================
# EXTRACT FILTERS
# =========================================================

def extract_where_clauses(cte_body):

    pattern = re.compile(
        r"\bWHERE\s+(.*?)"
        r"(?="
        r"\bGROUP\s+BY\b|"
        r"\bHAVING\b|"
        r"\bQUALIFY\b|"
        r"\bORDER\s+BY\b|"
        r"\bUNION\b|"
        r"$"
        r")",
        re.IGNORECASE | re.DOTALL
    )

    return [
        normalize_space(match.group(1))
        for match in pattern.finditer(cte_body)
    ]


# =========================================================
# EXTRACT GROUP BY
# =========================================================

def extract_group_bys(cte_body):

    pattern = re.compile(
        r"\bGROUP\s+BY\s+(.*?)"
        r"(?="
        r"\bHAVING\b|"
        r"\bQUALIFY\b|"
        r"\bORDER\s+BY\b|"
        r"\bUNION\b|"
        r"$"
        r")",
        re.IGNORECASE | re.DOTALL
    )

    return [
        normalize_space(match.group(1))
        for match in pattern.finditer(cte_body)
    ]


# =========================================================
# AGGREGATIONS
# =========================================================

def extract_aggregations(cte_body):

    pattern = re.compile(
        r"\b("
        r"SUM|COUNT|AVG|MIN|MAX"
        r")\s*\((.*?)\)",
        re.IGNORECASE | re.DOTALL
    )

    results = []

    for match in pattern.finditer(cte_body):

        function = match.group(1).upper()

        expression = normalize_space(
            match.group(2)
        )

        results.append(
            f"{function}({expression})"
        )

    return list(dict.fromkeys(results))


# =========================================================
# UNION DETECTION
# =========================================================

def extract_unions(cte_body):

    union_all = len(
        re.findall(
            r"\bUNION\s+ALL\b",
            cte_body,
            re.IGNORECASE
        )
    )

    union = len(
        re.findall(
            r"\bUNION\b(?!\s+ALL)",
            cte_body,
            re.IGNORECASE
        )
    )

    return union, union_all


# =========================================================
# IDENTIFY TABLE vs CTE DEPENDENCY
# =========================================================

def classify_dependencies(cte_blocks):

    cte_names = {
        cte["name"].lower()
        for cte in cte_blocks
    }

    dependencies = []

    for cte in cte_blocks:

        parent_cte = cte["name"]

        refs = extract_references(
            cte["body"]
        )

        seen = set()

        for ref in refs:

            obj = ref["object_name"]

            key = obj.lower()

            if key in cte_names:
                object_type = "CTE"
            else:
                object_type = "TABLE"

            dep_key = (
                parent_cte.lower(),
                key
            )

            if dep_key in seen:
                continue

            seen.add(dep_key)

            dependencies.append({
                "parent_cte": parent_cte,
                "depends_on": obj,
                "object_type": object_type,
                "alias": ref["alias"],
                "reference_type":
                    ref["source_type"]
            })

    return dependencies


# =========================================================
# FINAL SELECT DEPENDENCIES
# =========================================================

def extract_final_sql(sql, cte_blocks):

    if not cte_blocks:
        return sql

    end_position = cte_blocks[-1]["end"] + 1

    return sql[end_position:]


def extract_final_dependencies(
    final_sql,
    cte_names
):

    refs = extract_references(
        final_sql
    )

    results = []

    seen = set()

    cte_lower = {
        x.lower()
        for x in cte_names
    }

    for ref in refs:

        obj = ref["object_name"]

        key = obj.lower()

        if key in seen:
            continue

        seen.add(key)

        object_type = (
            "CTE"
            if key in cte_lower
            else "TABLE"
        )

        results.append({
            "parent_cte": "FINAL_SELECT",
            "depends_on": obj,
            "object_type": object_type,
            "alias": ref["alias"],
            "reference_type":
                ref["source_type"]
        })

    return results


# =========================================================
# CSV WRITER
# =========================================================

def write_csv(
    filename,
    headers,
    rows
):

    with open(
        filename,
        "w",
        newline="",
        encoding="utf-8"
    ) as f:

        writer = csv.DictWriter(
            f,
            fieldnames=headers
        )

        writer.writeheader()

        writer.writerows(rows)


# =========================================================
# MERMAID HELPERS
# =========================================================

def mermaid_id(text):
    """
    Create safe Mermaid node ID.
    """

    return re.sub(
        r"[^A-Za-z0-9_]",
        "_",
        text
    )


def mermaid_label(text):

    return (
        text
        .replace('"', "'")
        .replace("\n", " ")
    )


# =========================================================
# BUILD MERMAID DIAGRAM
# =========================================================

def build_mermaid(
    dependencies,
    joins
):

    lines = []

    lines.append(
        "flowchart TD"
    )

    lines.append("")

    all_ctes = set()

    all_tables = set()

    for dep in dependencies:

        if dep["parent_cte"] != "FINAL_SELECT":
            all_ctes.add(
                dep["parent_cte"]
            )

        if dep["object_type"] == "CTE":
            all_ctes.add(
                dep["depends_on"]
            )

        else:
            all_tables.add(
                dep["depends_on"]
            )

    # -------------------------------------
    # Tables subgraph
    # -------------------------------------

    lines.append(
        'subgraph TABLES["Source Tables"]'
    )

    for table in sorted(all_tables):

        node = "T_" + mermaid_id(
            table
        )

        label = mermaid_label(
            table
        )

        lines.append(
            f'    {node}["{label}"]'
        )

    lines.append("end")

    lines.append("")

    # -------------------------------------
    # CTE subgraph
    # -------------------------------------

    lines.append(
        'subgraph CTES["CTEs"]'
    )

    for cte in sorted(all_ctes):

        node = "C_" + mermaid_id(
            cte
        )

        label = mermaid_label(
            cte
        )

        lines.append(
            f'    {node}["{label}"]'
        )

    lines.append("end")

    lines.append("")

    lines.append(
        'FINAL["FINAL SELECT"]'
    )

    lines.append("")

    # -------------------------------------
    # Dependency arrows
    # -------------------------------------

    join_lookup = {}

    for join in joins:

        key = (
            join["parent_cte"].lower(),
            join["object_name"].lower()
        )

        if key not in join_lookup:
            join_lookup[key] = join[
                "join_type"
            ]

    for dep in dependencies:

        parent = dep["parent_cte"]

        source = dep["depends_on"]

        if dep["object_type"] == "CTE":

            source_node = (
                "C_" +
                mermaid_id(source)
            )

        else:

            source_node = (
                "T_" +
                mermaid_id(source)
            )

        if parent == "FINAL_SELECT":

            target_node = "FINAL"

        else:

            target_node = (
                "C_" +
                mermaid_id(parent)
            )

        join_key = (
            parent.lower(),
            source.lower()
        )

        relationship = ""

        if join_key in join_lookup:
            relationship = (
                join_lookup[join_key]
                + " JOIN"
            )

        elif dep["reference_type"] == "FROM":
            relationship = "FROM"

        if relationship:

            lines.append(
                f'{source_node} '
                f'-->|"{relationship}"| '
                f'{target_node}'
            )

        else:

            lines.append(
                f"{source_node} --> "
                f"{target_node}"
            )

    return "\n".join(lines)


# =========================================================
# MAIN
# =========================================================

def main():

    if len(sys.argv) < 2:

        print("")
        print("Usage:")
        print(
            "python analyze_sql_v2.py "
            "your_query.sql"
        )
        print("")

        return

    sql_file = Path(
        sys.argv[1]
    )

    if not sql_file.exists():

        print(
            f"File not found: "
            f"{sql_file}"
        )

        return

    print("")
    print(
        f"Reading: "
        f"{sql_file.name}"
    )

    sql = sql_file.read_text(
        encoding="utf-8",
        errors="ignore"
    )

    sql = clean_sql(
        sql
    )

    # -----------------------------------------
    # Extract CTEs
    # -----------------------------------------

    cte_blocks = extract_cte_blocks(
        sql
    )

    print(
        f"CTEs found: "
        f"{len(cte_blocks)}"
    )

    cte_names = [
        x["name"]
        for x in cte_blocks
    ]

    # -----------------------------------------
    # Dependencies
    # -----------------------------------------

    dependencies = classify_dependencies(
        cte_blocks
    )

    # -----------------------------------------
    # Final SELECT
    # -----------------------------------------

    final_sql = extract_final_sql(
        sql,
        cte_blocks
    )

    final_dependencies = (
        extract_final_dependencies(
            final_sql,
            cte_names
        )
    )

    dependencies.extend(
        final_dependencies
    )

    # -----------------------------------------
    # Detailed joins
    # -----------------------------------------

    all_joins = []

    for cte in cte_blocks:

        joins = extract_joins(
            cte["body"]
        )

        for join in joins:

            all_joins.append({
                "parent_cte":
                    cte["name"],
                "join_type":
                    join["join_type"],
                "object_name":
                    join["object_name"],
                "alias":
                    join["alias"],
                "condition":
                    join["condition"]
            })

    # -----------------------------------------
    # CTE analysis
    # -----------------------------------------

    cte_analysis = []

    for index, cte in enumerate(
        cte_blocks,
        start=1
    ):

        where_clauses = (
            extract_where_clauses(
                cte["body"]
            )
        )

        group_bys = (
            extract_group_bys(
                cte["body"]
            )
        )

        aggregations = (
            extract_aggregations(
                cte["body"]
            )
        )

        union_count, union_all_count = (
            extract_unions(
                cte["body"]
            )
        )

        cte_analysis.append({
            "sequence": index,
            "cte_name":
                cte["name"],
            "where_count":
                len(where_clauses),
            "group_by_count":
                len(group_bys),
            "aggregation_count":
                len(aggregations),
            "union_count":
                union_count,
            "union_all_count":
                union_all_count
        })

    # -----------------------------------------
    # Output directory
    # -----------------------------------------

    output_dir = (
        sql_file.parent /
        "sql_analysis_v2"
    )

    output_dir.mkdir(
        exist_ok=True
    )

    # -----------------------------------------
    # Write CTE list
    # -----------------------------------------

    write_csv(
        output_dir /
        "ctes.csv",
        [
            "sequence",
            "cte_name"
        ],
        [
            {
                "sequence": i + 1,
                "cte_name": name
            }
            for i, name
            in enumerate(cte_names)
        ]
    )

    # -----------------------------------------
    # Write dependencies
    # -----------------------------------------

    write_csv(
        output_dir /
        "cte_dependencies.csv",
        [
            "parent_cte",
            "depends_on",
            "object_type",
            "alias",
            "reference_type"
        ],
        dependencies
    )

    # -----------------------------------------
    # Write joins
    # -----------------------------------------

    write_csv(
        output_dir /
        "joins_by_cte.csv",
        [
            "parent_cte",
            "join_type",
            "object_name",
            "alias",
            "condition"
        ],
        all_joins
    )

    # -----------------------------------------
    # Write CTE analysis
    # -----------------------------------------

    write_csv(
        output_dir /
        "cte_analysis.csv",
        [
            "sequence",
            "cte_name",
            "where_count",
            "group_by_count",
            "aggregation_count",
            "union_count",
            "union_all_count"
        ],
        cte_analysis
    )

    # -----------------------------------------
    # Split each CTE into separate SQL file
    # -----------------------------------------

    cte_folder = (
        output_dir /
        "cte_sql"
    )

    cte_folder.mkdir(
        exist_ok=True
    )

    for index, cte in enumerate(
        cte_blocks,
        start=1
    ):

        safe_name = re.sub(
            r"[^A-Za-z0-9_-]",
            "_",
            cte["name"]
        )

        filename = (
            cte_folder /
            f"{index:02d}_"
            f"{safe_name}.sql"
        )

        filename.write_text(
            cte["body"],
            encoding="utf-8"
        )

    # -----------------------------------------
    # Mermaid
    # -----------------------------------------

    diagram = build_mermaid(
        dependencies,
        all_joins
    )

    (
        output_dir /
        "flow_diagram.mmd"
    ).write_text(
        diagram,
        encoding="utf-8"
    )

    # -----------------------------------------
    # Summary TXT
    # -----------------------------------------

    summary_file = (
        output_dir /
        "summary.txt"
    )

    with summary_file.open(
        "w",
        encoding="utf-8"
    ) as f:

        f.write(
            "SQL ANALYSIS VERSION 2\n"
        )

        f.write(
            "======================\n\n"
        )

        f.write(
            f"CTEs found: "
            f"{len(cte_blocks)}\n"
        )

        f.write(
            f"Dependencies found: "
            f"{len(dependencies)}\n"
        )

        f.write(
            f"Joins found: "
            f"{len(all_joins)}\n\n"
        )

        f.write(
            "CTE ORDER\n"
        )

        f.write(
            "---------\n"
        )

        for index, name in enumerate(
            cte_names,
            start=1
        ):

            f.write(
                f"{index}. {name}\n"
            )

        f.write(
            "\n\nDEPENDENCIES\n"
        )

        f.write(
            "------------\n"
        )

        for dep in dependencies:

            f.write(
                f"{dep['depends_on']} "
                f"--> "
                f"{dep['parent_cte']} "
                f"[{dep['object_type']}]\n"
            )

    # -----------------------------------------
    # Done
    # -----------------------------------------

    print("")
    print(
        "Analysis completed."
    )

    print("")
    print(
        f"Output folder:"
    )

    print(
        output_dir
    )

    print("")
    print(
        "Files generated:"
    )

    print(
        "1. ctes.csv"
    )

    print(
        "2. cte_dependencies.csv"
    )

    print(
        "3. joins_by_cte.csv"
    )

    print(
        "4. cte_analysis.csv"
    )

    print(
        "5. flow_diagram.mmd"
    )

    print(
        "6. summary.txt"
    )

    print(
        "7. cte_sql folder"
    )

    print("")


if __name__ == "__main__":
    main()
